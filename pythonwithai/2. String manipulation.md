## Parse Daily Sales Into Lists

#### - `daily_sales` is structured as a single long string where:
#### - Individual transactions are separated by a comma ','
#### - Data points within a transaction are separated by ';,;'

```python
print(daily_sales)

# Step 2: Replace ';,;' with '+' to avoid splitting issues when splitting by comma
daily_sales_replaced = daily_sales.replace(";,;", "+")

# Step 3: Split the full string into a list of individual transaction strings
daily_transactions = daily_sales_replaced.split(",")

# Step 4: Print daily_transactions to check output
print(daily_transactions)

# Step 5: Define an empty list to store split data points
daily_transactions_split = []

# Step 6: Iterate through daily_transactions and split each transaction by '+'
for transaction in daily_transactions:
    daily_transactions_split.append(transaction.split("+"))

# Step 7: Print daily_transactions_split to inspect the list of lists
print(daily_transactions_split)

# Step 8: Define an empty list for cleaned transaction data
transactions_clean = []

# Iterate over each transaction and strip whitespace from each data point
for transaction in daily_transactions_split:
    transaction_data = []
    for data_point in transaction:
        transaction_data.append(data_point.strip())
    transactions_clean.append(transaction_data)

# Step 9: Print transactions_clean to verify removal of unwanted spaces
print(transactions_clean)

# Step 10: Create empty lists for each attributes
customers = []
sales = []
thread_sold = []

# Step 11: Iterate through cleaned transactions and append data to respective lists
for transaction in transactions_clean:
    customers.append(transaction[0])
    sales.append(transaction[1])
    thread_sold.append(transaction[2])

# Step 12: Print all three lists to confirm correctness
print(customers)
print(sales)
print(thread_sold)

# Step 13: Define total_sales initialized to 0
total_sales = 0

# Step 14: Strip '$', convert string to float, and sum values
for sale in sales:
    total_sales += float(sale.strip("$"))

# Step 15: Print total sales formatted or raw
print(total_sales)

# Step 16: Inspect thread_sold
print(thread_sold)

# Step 17: Define an empty list for individual thread colors
thread_sold_split = []

# Step 18: Handle single vs. multiple colors separated by '&'
for item in thread_sold:
    for color in item.split("&"):
        thread_sold_split.append(color.strip())

# Step 19: Define color_count function
def color_count(color):
    count = 0
    for thread in thread_sold_split:
        if thread == color:
            count += 1
    return count

# Step 20: Test function (returns 28)
print(color_count("white"))

# Step 21: Define available thread colors
colors = ["red", "yellow", "green", "white", "black", "blue", "purple"]

# Step 22: Iterate through colors and print summary using .format()
for color in colors:
    print(
        "Thread Shed sold {} threads of {} thread today.".format(
            color_count(color), color
        )
    )
```
